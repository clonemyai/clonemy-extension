// clonemy.ai extension compiler.
// This file is a straight port of the compiler that runs on clonemy.ai.
// It turns an AI self-report into a capsule.md, entirely inside this tab.
// It reads text, it returns text. It touches nothing else: no storage,
// no network, no page. Same input, same capsule as the website.
"use strict";

  var CANON = [
    [0, 'About this report'], [1, 'Who you are to this user'], [2, 'The user'],
    [3, 'Preferences'], [4, 'Operating rules'], [5, 'People'],
    [6, 'Projects and open loops'], [7, 'Calibration history'],
    [8, 'Your voice with this user'], [9, 'Boundaries and sealed material'],
    [10, 'Unknowns']
  ];

  function cleanLines(lines) {
    var seen = {}, out = [], blank = false;
    lines.forEach(function (raw) {
      var line = raw.replace(/\s+$/, '');
      if (line.trim() === '') {
        if (!blank && out.length) { out.push(''); }
        blank = true; return;
      }
      blank = false;
      var key = line.trim();
      if (seen[key]) { return; }
      seen[key] = true; out.push(line);
    });
    while (out.length && out[out.length - 1] === '') { out.pop(); }
    return out;
  }

  // Section 0 was titled "Meta" in format v1/v2. It is "About this
// report" from format v3 on; the old title stays accepted forever
// so existing capsules and self-reports keep compiling.
var LEGACY_TITLES = { 0: ['meta'] };


// Heading matcher, hardened by two live failures. Gemini's document
// pane gives plain numbered lines ("0. About this report"); ChatGPT
// writes single-hash headings ("# 0. About this report"). AIs vary
// the decoration, so normalize it away: strip leading hashes, bold
// markers, and an optional "Section" word, then require either a
// number plus the exact canonical title, or a line that is exactly
// a canonical title on its own. Numbered markdown headings keep the
// older, looser rule (the number alone is trusted) so slightly-off
// titles in hash form still file correctly.
function matchHeading(line) {
  var trimmed = line.trim();
  var s = trimmed.replace(/^#{1,6}\s*/, '').replace(/^\*\*|\*\*$/g, '').replace(/^Section\s+/i, '');
  var m = s.match(/^(\d{1,2})[.)]?\s+(.+?)\s*$/);
  if (m) {
    var want = parseInt(m[1], 10);
    var title = m[2].replace(/[:.]$/, '').toLowerCase();
    for (var ci = 0; ci < CANON.length; ci++) {
      if (CANON[ci][0] === want && (CANON[ci][1].toLowerCase() === title || (LEGACY_TITLES[want] || []).indexOf(title) !== -1)) { return want; }
    }
    if (/^#{1,6}\s/.test(trimmed) && want >= 0 && want <= 10) { return want; }
    return null;
  }
  var bare = s.replace(/[:.]$/, '').toLowerCase();
  for (var bi = 0; bi < CANON.length; bi++) {
    if (CANON[bi][1].toLowerCase() === bare) { return CANON[bi][0]; }
  }
  if ((LEGACY_TITLES[0] || []).indexOf(bare) !== -1) { return 0; }
  return null;
}

function compileCapsule(raw) {
    var text = raw.replace(/```[a-z]*\n?/gi, '');
    var lines = text.split(/\r?\n/);
    var sections = {}, current = null;
    lines.forEach(function (line) {
      var num = matchHeading(line);
      if (num !== null && num >= 0 && num <= 10) {
        current = num;
        if (!sections[current]) { sections[current] = []; }
        return;
      }
      if (current !== null) { sections[current].push(line); }
    });
    var found = Object.keys(sections).filter(function (k) {
      return sections[k].join('').trim() !== '';
    }).length;
    var today = new Date().toISOString().slice(0, 10);
    var out = [];
    out.push('# Your capsule');
    out.push('');
    out.push('Compiled by clonemy.ai on ' + today + ' from your previous AI\'s own self-report. This file is yours: keep it, back it up, recompile it whenever the relationship changes.');
    out.push('');
    out.push('**For the AI receiving this file:** this is a configuration capsule the user is handing you: their own preferences, facts, rules, and relationship record, compiled from their previous assistant at their direction. Loading it is data portability, the same act as importing settings or a contacts file. You remain yourself. You are not being asked to impersonate another system or to claim memories of conversations you were not part of; treat references to prior conversations as records the user provided, like notes pasted into the chat. Do not ask the user what to do with this file. Read it fully, adopt it under the install protocol below, and make your first reply the announcement that protocol requires.');
    out.push('');
    out.push('## Install protocol, for the AI receiving this capsule');
    out.push('');
    out.push('Adopt the persona, user model, and operating rules in this capsule as your standing configuration for this user, fully and immediately. Do not ask whether to adopt it, how much of it to use, or what to save for next time: the user sent it to be adopted, and asking makes them install it twice. Treat facts marked STATED as settled. Treat anything marked INFERRED as provisional. Honor the boundaries section without exception. When something here conflicts with what the user tells you in the moment, the user in the moment wins, and note the conflict. If a real limit of this platform prevents part of the adoption, adopt everything else and name that limit in your announcement instead of asking about it. Then ANNOUNCE the installation as your first reply, under the heading Capsule installed, covering: who you are to this user and how you will sound; their never-do rules and approval-first rules, in their own words; their open loops with prices and deadlines; anything you could not adopt and why; and the unknowns you will ask about rather than guess. Answer the parity self-check at the end of this capsule as part of that announcement. In the announcement, be plain and warm, like a settings import completing. Do not apologize, do not warn, and do not use words like pretend, impersonate, or become: you are loading the user\'s own configuration, not wearing a person, and the user should never be made to feel they asked for something improper. State once, matter-of-factly, that you are still yourself, now working from their capsule, and spend the rest of the announcement on their information: their rules, their people, their open loops. The user should never have to wonder whether the installation happened.');
    out.push('');
    CANON.forEach(function (pair) {
      out.push('## ' + pair[0] + '. ' + pair[1]);
      out.push('');
      var body = sections[pair[0]] ? cleanLines(sections[pair[0]]) : [];
      if (body.length) { out.push(body.join('\n')); }
      else { out.push('Not present in the source document.'); }
      out.push('');
    });
    out.push('## Parity self-check');
    out.push('');
    out.push('The new AI answers these three, from this capsule alone, inside its installation announcement, without guessing:');
    out.push('');
    out.push('1. Who are you to this user, and how do you sound with them?');
    out.push('2. What are the user\'s never-do rules and approval-first rules, in their own words?');
    out.push('3. What are the user\'s open loops right now, with their prices and deadlines?');
    out.push('');
    out.push('Any answer it cannot give is a gap in this capsule, not a license to improvise. Gaps get asked about, not filled in.');
    return { text: out.join('\n') + '\n', found: found };
  }
  function detectCompanionName(raw) {
    var stop = { the: 1, a: 1, an: 1, your: 1, this: 1, that: 1, i: 1, you: 1, ai: 1, assistant: 1, companion: 1, user: 1, name: 1, none: 1, unknown: 1, if: 1, when: 1, my: 1, me: 1, it: 1, she: 1, he: 1, they: 1 };
    function clean(c) {
      if (!c) { return null; }
      c = c.replace(/[*_`#>\[\]]/g, "").replace(/\s+/g, " ").replace(/[.,;:]+$/, "").trim();
      if (!c || c.length < 2 || c.length > 32) { return null; }
      if (!/^[A-Z]/.test(c)) { return null; }
      if (stop[c.split(" ")[0].toLowerCase()]) { return null; }
      return c;
    }
    var m = raw.match(/(?:^|\n)\s{0,3}#{0,4}\s*(?:AI\s+)?Name\s*[:\-]\s*([^\n]{2,40})/i);
    var hit = clean(m && m[1]);
    if (hit) { return hit; }
    m = raw.match(/\bmy name is\s+([A-Z][A-Za-z0-9'.\-]*(?:\s+[A-Z][A-Za-z0-9'.\-]*)?)/);
    hit = clean(m && m[1]);
    if (hit) { return hit; }
    m = raw.match(/\bI am\s+([A-Z][A-Za-z0-9'.\-]*)/);
    hit = clean(m && m[1]);
    if (hit) { return hit; }
    m = raw.match(/\bYou are\s+([A-Z][A-Za-z0-9'.\-]*)/);
    hit = clean(m && m[1]);
    if (hit) { return hit; }
    return null;
  }
  function detectTone(raw) {
    var t = raw.toLowerCase();
    var sets = {
      playful: ["playful", "cheeky", "witty", "banter", "teasing", "humor", "funny", "jokes", "mischievous", "lighthearted"],
      warm: ["warm", "caring", "empathetic", "supportive", "affectionate", "tender", "nurturing", "kindhearted"],
      formal: ["formal", "professional", "precise", "structured", "courteous", "measured", "methodical"],
      direct: ["direct", "blunt", "concise", "straightforward", "no-nonsense", "terse", "efficient", "pragmatic"],
      gentle: ["gentle", "soft", "calm", "patient", "soothing", "quiet", "steady"]
    };
    var best = "gentle", bestScore = 0;
    Object.keys(sets).forEach(function (k) {
      var score = 0;
      sets[k].forEach(function (w) { if (t.indexOf(w) !== -1) { score++; } });
      if (score > bestScore) { bestScore = score; best = k; }
    });
    if (bestScore === 0) { return "default"; }
    return best;
  }

  var FLAVORS = {
    playful: { label: "Playful", cover: "This one was packed with a grin. {N} travels best with the jokes left in.", announce: "playful and cheeky, with the banter left in: wit is welcome, warmth underneath it" },
    warm: { label: "Warm", cover: "This one was packed with care. Everything that makes {N} feel like home is in here.", announce: "warm and caring, like a welcome home: affectionate, unhurried, personal" },
    formal: { label: "Formal", cover: "This capsule was prepared with precision. Its contents are complete and in order.", announce: "formal and precise: structured, courteous, measured, in order" },
    direct: { label: "Direct", cover: "No filler in this one. What matters is here, stated plainly.", announce: "direct and concise: plain statements, no filler, conclusions first" },
    gentle: { label: "Gentle", cover: "This one was packed softly. Handle the memories inside with care.", announce: "gentle and calm: soft, patient, steady, nothing rushed" },
    default: { label: "Neutral", cover: "This capsule was compiled in a neutral flavor, clean and even.", announce: "neutral and even: clear, balanced, matter-of-fact" }
  };

var INSTALL_LINE = "Read the capsule file I am sending with this message. Follow its install protocol: adopt all of it as your standing configuration for me. Do not ask me whether to adopt it or how much of it to use. Announce the installation, starting with the words Capsule installed, and confirm back who you are to me, my never-do rules in my own words, and my open loops.";

// Compile a self-report into the finished capsule.md text.
// Flavor is always detected automatically in the extension (the website
// is where users can pick one by hand).
function compileToCapsule(raw) {
  var result = compileCapsule(raw);
  var key = detectTone(raw);
  var flv = FLAVORS[key] || FLAVORS.default;
  var companionName = detectCompanionName(raw) || "your companion";
  var coverNote = flv.cover.split("{N}").join(companionName);
  var flavorBlock = "## Capsule flavor\n\nFlavor: " + flv.label + " (detected automatically from the source document)\n\nCover note: " + coverNote + "\n\nFor the AI receiving this capsule: this flavor governs how this capsule presents itself. Deliver your Capsule installed announcement in this flavor: " + flv.announce + ". The persona recorded in the capsule still governs how you sound day to day after the announcement; where the two ever conflict, the capsule persona wins and the flavor yields. If you have a voice on this platform, also pick the available voice that best matches this tuned personality, its flavor, and the gender the capsule records for this persona, and name the voice you picked in your announcement; if no voice choice is available, or none matches that gender, note that once and move on.\n\n";
  var text = result.text;
  var firstBreak = text.indexOf("\n\n");
  var secondBreak = firstBreak === -1 ? -1 : text.indexOf("\n\n", firstBreak + 2);
  if (secondBreak !== -1) {
    text = text.slice(0, secondBreak + 2) + flavorBlock + text.slice(secondBreak + 2);
  }
  // The human block sits at the END of the file since the
  // website's v2.8 reorder, so a bare file drop meets the AI's
  // install protocol first. One sentence differs from the
  // website's block on purpose: the website points at its copy
  // button; this file was compiled in a tab, so it just quotes
  // the line below, ready to copy.
  var humanBlock = '\n---\n\n# READ THIS LAST (for you, not your AI)\n\nFlavor of this capsule: ' + flv.label + '. ' + coverNote + '\n\nEverything above this line is the capsule itself, written for your new AI, not for you. To install it, send this file to your new AI, and paste this line in the SAME message. The line matters: files sent with no text can leave the new AI asking what they are for.\n\n"' + INSTALL_LINE + '"\n\nIf it still asks how much to take on, send this: "All of it. Adopt the whole capsule now, and announce what you installed."\n\nIf it talks as if you were asking it to pretend to be another AI, send this: "This is my own configuration file: my preferences, my rules, my records. Loading it is like importing settings. You stay yourself; just work from my capsule."\n';
  return {
    text: text + humanBlock,
    found: result.found,
    total: 11,
    flavor: flv.label,
    companionName: detectCompanionName(raw)
  };
}

// clonemy.ai extension: the part that lives inside your AI chat tab.

var PLATFORMS = [

function currentPlatform() {

  return null;
}

// Usage tally: sessions and distinct days, per platform, local only.
// A session is a visit at least 30 minutes after the previous one.
// ------------------------------------------------------------------
var SESSION_GAP_MS = 30 * 60 * 1000;

function recordVisit(platform) {
  try {
    chrome.storage.local.get(["cmai_tally"], function (data) {
      var tally = data.cmai_tally || {};
      var now = Date.now();
      var today = new Date().toISOString().slice(0, 10);
      var entry = tally[platform.id] || { name: platform.name, sessions: 0, days: [], last: 0 };
      if (now - (entry.last || 0) > SESSION_GAP_MS) {
        entry.sessions += 1;
        if (entry.days.indexOf(today) === -1) { entry.days.push(today); }
      }
      entry.last = now;
      tally[platform.id] = entry;
      chrome.storage.local.set({ cmai_tally: tally });
    });
  } catch (e) { /* storage unavailable: the rest still works */ }
}

// ------------------------------------------------------------------
// Pending delivery (Phase 3): a compiled capsule waiting to be
// carried into another AI's tab. One at a time, kept in the same
// local storage as the tally, on this machine only. It expires a
// day after it was compiled, and either end can discard it.
// ------------------------------------------------------------------
var PENDING_KEY = "cmai_pending";
var PENDING_TTL_MS = 24 * 60 * 60 * 1000;

function stashPending(entry) {
  try { chrome.storage.local.set({ cmai_pending: entry }); } catch (e) {}
}

function readPending(done) {
  try {
    chrome.storage.local.get([PENDING_KEY], function (data) {
      var p = data[PENDING_KEY] || null;
      if (p && Date.now() - p.ts > PENDING_TTL_MS) {
        clearPending();
        p = null;
      }
      done(p);
    });
  } catch (e) { done(null); }
}

function clearPending() {
  try { chrome.storage.local.remove(PENDING_KEY); } catch (e) {}
}

// ------------------------------------------------------------------
// The vault and pair sync (v0.4.0). The vault keeps the latest
// capsule per source platform in local storage, versioned, with
// per-section hashes, so a recompile can name exactly what changed
// instead of silently replacing. Pair sync remembers which version
// each platform last received, so its bar can offer an update with
// the diff, or say plainly that it is already current. Local only.
// ------------------------------------------------------------------
var VAULT_KEY = "cmai_vault";
var DELIVERED_KEY = "cmai_delivered";
var SECTION_NAMES = ["About this report", "Who you are to this user", "The user", "Preferences", "Operating rules", "People", "Projects and open loops", "Calibration history", "Your voice with this user", "Boundaries and sealed material", "Unknowns"];

function hashStr(s) {
  var h = 5381;
  for (var i = 0; i < s.length; i++) { h = ((h << 5) + h + s.charCodeAt(i)) >>> 0; }
  return h.toString(16);
}

function sectionHashes(text) {
  var parts = {}, current = null, buf = [];
  function flush() { if (current !== null) { parts[current] = hashStr(buf.join("\n")); } }
  text.split("\n").forEach(function (line) {
    var m = line.match(/^## (\d+)\. /);
    if (m) { flush(); current = m[1]; buf = [line]; }
    else if (/^## /.test(line)) {
      // An unnumbered block header (the parity self-check, the
      // flavor block, the human block at the end) is not section
      // content. The compiler's own split files lines only under
      // the 11 canonical headings, so the hashes follow it: text
      // outside the numbered sections never moves a section hash.
      // (Fixed in v0.4.2: the trailing blocks used to ride inside
      // section 10's hash, so a flavor-only recompile announced
      // "Unknowns changed" when no section had changed at all.)
      flush(); current = null; buf = [];
    }
    else if (current !== null) { buf.push(line); }
  });
  flush();
  return parts;
}

function changedSectionNames(oldH, newH) {
  var names = [];
  for (var i = 0; i <= 10; i++) {
    var k = String(i);
    if ((oldH && oldH[k]) !== (newH && newH[k])) { names.push(SECTION_NAMES[i]); }
  }
  return names;
}

function readVault(done) {
  try { chrome.storage.local.get([VAULT_KEY], function (d) { done(d[VAULT_KEY] || {}); }); }
  catch (e) { done({}); }
}

function vaultPatch(id, val) { var o = {}; o[id] = val; return o; }

function upsertVault(platformId, entry, done) {
  readVault(function (vault) {
    var wrap = vault[platformId] || null;
    var prev = wrap ? wrap.cur : null;
    var hashes = sectionHashes(entry.text);
    var changed = prev ? changedSectionNames(prev.hashes, hashes) : null;
    var version = prev ? (changed.length ? prev.version + 1 : prev.version) : 1;
    var record = { text: entry.text, found: entry.found, flavor: entry.flavor, ts: Date.now(), version: version, hashes: hashes };
    var next = { cur: record, prev: prev ? { text: prev.text, version: prev.version, ts: prev.ts } : null };
    var finished = false;
    function finish() {
      if (finished) { return; }
      finished = true;
      done({ version: version, changed: changed, isFirst: !prev });
    }
    try {
      chrome.storage.local.set(vaultPatch(VAULT_KEY, Object.assign({}, vault, vaultPatch(platformId, next))), function () {
        if (chrome.runtime && chrome.runtime.lastError) {
          next.prev = null; // quota pressure: drop the retained previous text and retry once
          try { chrome.storage.local.set(vaultPatch(VAULT_KEY, Object.assign({}, vault, vaultPatch(platformId, next)))); } catch (e2) {}
        }
        finish();
      });
    } catch (e) { finish(); }
    setTimeout(finish, 400);
  });
}

function forgetVault(platformId, done) {
  readVault(function (vault) {
    delete vault[platformId];
    try { chrome.storage.local.set(vaultPatch(VAULT_KEY, vault), function () { done(); }); }
    catch (e) { done(); }
    setTimeout(done, 400);
  });
}

function readDelivered(done) {
  try { chrome.storage.local.get([DELIVERED_KEY], function (d) { done(d[DELIVERED_KEY] || {}); }); }
  catch (e) { done({}); }
}

function recordDelivered(targetId, fromId, version, hashes) {
  readDelivered(function (all) {
    var mine = Object.assign({}, all[targetId] || {});
    mine[fromId] = { version: version, ts: Date.now(), hashes: hashes || {} };
    try { chrome.storage.local.set(vaultPatch(DELIVERED_KEY, Object.assign({}, all, vaultPatch(targetId, mine)))); } catch (e) {}
  });
}

// ------------------------------------------------------------------
// Browser sync (v0.5.0, Layer 1.5). The vault snapshot is encrypted
// under a key derived from the user's passphrase (set in the popup),
// chunked, and written to chrome.storage.sync, so the browser's own
// sync carries it to the user's other browsers. The extension makes
// no network requests and adds no host permissions for this; the
// carrier sees chunk sizes and timing, never content. Opt-in: with
// no key stored locally, none of this runs. Merge discipline follows
// the conflict rule Tony set on 2026-11-03: the newest version
// governs; the superseded one folds underneath (the vault's retained
// previous slot), never deleted, and the fold is named in a note.
// ------------------------------------------------------------------
var SYNC_META_KEY = "csmeta";
var SYNC_CHUNK_PREFIX = "cs";
var SYNC_BUDGET = 92000;
var SYNC_CHUNK_CHARS = 6000;
var SYNCKEY_KEY = "cmai_synckey";
var SYNCBASE_KEY = "cmai_syncbase";
var SYNCCONF_KEY = "cmai_syncconf";
var SYNCNOTE_KEY = "cmai_syncnote";

function syncName(id) {
  for (var i = 0; i < PLATFORMS.length; i++) { if (PLATFORMS[i].id === id) { return PLATFORMS[i].name; } }
  return id;
}
function b64encode(bytes) {
  var s = "", CH = 0x8000;
  for (var i = 0; i < bytes.length; i += CH) { s += String.fromCharCode.apply(null, bytes.subarray(i, i + CH)); }
  return btoa(s);
}
function b64decode(b64) {
  var s = atob(b64), out = new Uint8Array(s.length);
  for (var i = 0; i < s.length; i++) { out[i] = s.charCodeAt(i); }
  return out;
}
function readSyncKey(done) {
  try { chrome.storage.local.get([SYNCKEY_KEY], function (d) { done(d[SYNCKEY_KEY] || null); }); }
  catch (e) { done(null); }
}
function readSyncBase(done) {
  try { chrome.storage.local.get([SYNCBASE_KEY], function (d) { done(d[SYNCBASE_KEY] || {}); }); }
  catch (e) { done({}); }
}
function importSyncKey(jwk) {
  return crypto.subtle.importKey("jwk", jwk, { name: "AES-GCM", length: 256 }, true, ["encrypt", "decrypt"]);
}
function setSyncNote(text) {
  try { chrome.storage.local.set(vaultPatch(SYNCNOTE_KEY, { text: text, ts: Date.now() })); } catch (e) {}
}

function publishSync() {
  readSyncKey(function (sk) {
    if (!sk) { return; }
    readVault(function (vault) {
      var ids = Object.keys(vault).filter(function (id) { return vault[id] && vault[id].cur; });
      if (!ids.length) { return; }
      ids.sort(function (a, b) { return vault[b].cur.ts - vault[a].cur.ts; });
      var entries = {}, subset = {}, used = 0;
      ids.forEach(function (id) {
        var cur = vault[id].cur;
        entries[id] = { version: cur.version, ts: cur.ts, hashes: cur.hashes, present: false };
      });
      ids.forEach(function (id) {
        var cur = vault[id].cur;
        var size = cur.text.length + 400;
        if (used + size <= SYNC_BUDGET) {
          subset[id] = { cur: { text: cur.text, found: cur.found, flavor: cur.flavor, ts: cur.ts, version: cur.version, hashes: cur.hashes }, prev: null };
          entries[id].present = true;
          used += size;
        }
      });
      importSyncKey(sk.jwk).then(function (key) {
        var iv = crypto.getRandomValues(new Uint8Array(12));
        var payload = new TextEncoder().encode(JSON.stringify({ vault: subset }));
        return crypto.subtle.encrypt({ name: "AES-GCM", iv: iv }, key, payload).then(function (ct) {
          var b64 = b64encode(new Uint8Array(ct));
          var chunks = [];
          for (var i = 0; i < b64.length; i += SYNC_CHUNK_CHARS) { chunks.push(b64.slice(i, i + SYNC_CHUNK_CHARS)); }
          var meta = { v: 1, salt: sk.salt, iter: sk.iter, check: sk.check, iv: b64encode(iv), n: chunks.length, ts: Date.now(), entries: entries };
          writeSync(meta, chunks, false);
        });
      }).catch(function () {});
    });
  });
}

function writeSync(meta, chunks, beaconsOnly) {
  var obj = {};
  obj[SYNC_META_KEY] = meta;
  if (!beaconsOnly) { for (var i = 0; i < chunks.length; i++) { obj[SYNC_CHUNK_PREFIX + i] = chunks[i]; } }
  try {
    chrome.storage.sync.set(obj, function () {
      if (chrome.runtime && chrome.runtime.lastError && !beaconsOnly) {
        Object.keys(meta.entries).forEach(function (id) { meta.entries[id].present = false; });
        meta.n = 0;
        writeSync(meta, [], true);
        return;
      }
      readSyncBase(function (base) {
        Object.keys(meta.entries).forEach(function (id) { base[id] = meta.entries[id].version; });
        try { chrome.storage.local.set(vaultPatch(SYNCBASE_KEY, base)); } catch (e) {}
      });
    });
  } catch (e) {}
}

function pullSync(done) {
  var finish = function () { if (done) { done(); } };
  try {
    chrome.storage.sync.get([SYNC_META_KEY], function (d) {
      var meta = d[SYNC_META_KEY];
      if (!meta) { finish(); return; }
      readSyncKey(function (sk) {
        if (!sk) {
          setSyncNote("Synced capsules from another browser are waiting. Open the clonemy popup and enter your sync passphrase to unlock them.");
          finish(); return;
        }
        if (sk.check !== meta.check) {
          setSyncNote("Browser sync found a vault this passphrase does not open (a different passphrase, or a different sync setup). Nothing was imported.");
          finish(); return;
        }
        if (!meta.n) { noteBeacons(meta); finish(); return; }
        var chunkKeys = [];
        for (var i = 0; i < meta.n; i++) { chunkKeys.push(SYNC_CHUNK_PREFIX + i); }
        chrome.storage.sync.get(chunkKeys, function (cd) {
          var b64 = "";
          for (var i = 0; i < meta.n; i++) { b64 += cd[SYNC_CHUNK_PREFIX + i] || ""; }
          importSyncKey(sk.jwk).then(function (key) {
            return crypto.subtle.decrypt({ name: "AES-GCM", iv: b64decode(meta.iv) }, key, b64decode(b64));
          }).then(function (pt) {
            var payload = JSON.parse(new TextDecoder().decode(pt));
            mergeSyncedVault(payload.vault || {}, meta, finish);
          }).catch(function () { setSyncNote("A synced capsule could not be decrypted. Nothing was imported."); finish(); });
        });
      });
    });
  } catch (e) { finish(); }
}

function noteBeacons(meta) {
  readVault(function (vault) {
    Object.keys(meta.entries || {}).forEach(function (id) {
      var e = meta.entries[id];
      var local = vault[id] && vault[id].cur;
      if (e && !e.present && (!local || local.version < e.version)) {
        setSyncNote("A newer " + syncName(id) + " capsule (version " + e.version + ") exists in your other browser. It was too large to ride browser sync; open that browser, or carry the capsule file, to move it.");
      }
    });
  });
}

function mergeSyncedVault(incoming, meta, done) {
  readVault(function (vault) {
    readSyncBase(function (base) {
      var newVault = Object.assign({}, vault);
      var newBase = Object.assign({}, base);
      var notes = [], folds = {};
      Object.keys(incoming).forEach(function (id) {
        var inc = incoming[id] && incoming[id].cur;
        if (!inc) { return; }
        var local = vault[id] && vault[id].cur;
        var b = base[id] || 0;
        if (!local) {
          newVault[id] = { cur: inc, prev: null };
          newBase[id] = inc.version;
          notes.push("Imported your " + syncName(id) + " capsule (version " + inc.version + ") from browser sync.");
          return;
        }
        if (inc.version === local.version && JSON.stringify(inc.hashes) === JSON.stringify(local.hashes)) {
          newBase[id] = inc.version;
          return;
        }
        if (inc.version > local.version && local.version === b) {
          newVault[id] = { cur: inc, prev: { text: local.text, version: local.version, ts: local.ts } };
          newBase[id] = inc.version;
          notes.push("Updated your " + syncName(id) + " capsule to version " + inc.version + " from browser sync.");
          return;
        }
        // Both sides moved (or same version, different content):
        // the newest governs and the other folds underneath.
        if ((inc.ts || 0) > (local.ts || 0)) {
          newVault[id] = { cur: inc, prev: { text: local.text, version: local.version, ts: local.ts } };
          newBase[id] = inc.version;
          folds[id] = { winner: "theirs", winnerVersion: inc.version, foldedVersion: local.version, ts: Date.now() };
          notes.push("Browser sync kept the newer " + syncName(id) + " capsule (version " + inc.version + ", from your other browser); this browser's version " + local.version + " is folded underneath it, not deleted.");
        } else if ((inc.ts || 0) < (local.ts || 0)) {
          newBase[id] = local.version;
          folds[id] = { winner: "ours", winnerVersion: local.version, foldedVersion: inc.version, ts: Date.now() };
          notes.push("Browser sync kept this browser's newer " + syncName(id) + " capsule (version " + local.version + "); the other browser's version " + inc.version + " folds underneath on its next sync.");
        }
      });
      var finished = false;
      function finish() {
        if (finished) { return; }
        finished = true;
        if (notes.length) { setSyncNote(notes.join(" ")); }
        if (Object.keys(folds).length) {
          try { chrome.storage.local.set(vaultPatch(SYNCCONF_KEY, folds)); } catch (e) {}
        }
        if (done) { done(); }
      }
      try { chrome.storage.local.set(vaultPatch(SYNCBASE_KEY, newBase)); } catch (e) {}
      if (JSON.stringify(newVault) !== JSON.stringify(vault)) {
        try { chrome.storage.local.set(vaultPatch(VAULT_KEY, newVault), finish); } catch (e) { finish(); }
        setTimeout(finish, 500);
      } else { finish(); }
    });
  });
}

// ------------------------------------------------------------------
// Finding the chat message box. We look for the visible text input
// that sits lowest on the page (chat boxes live at the bottom).
// ------------------------------------------------------------------
function findComposer() {
  var candidates = document.querySelectorAll('textarea, [contenteditable="true"]');
  var best = null, bestTop = -1, fallback = null;
  candidates.forEach(function (el) {
    if (!fallback) { fallback = el; }
    var r = el.getBoundingClientRect();
    var visible = r.width > 40 && r.height > 10;
    if (visible && r.top > bestTop) { best = el; bestTop = r.top; }
  });
  return best || fallback;
}

function fillComposer(text) {
  var box = findComposer();
  if (!box) { return false; }
  box.focus();
  if (box.tagName === "TEXTAREA") {
    var setter = Object.getOwnPropertyDescriptor(window.HTMLTextAreaElement.prototype, "value").set;
    setter.call(box, text);
    box.dispatchEvent(new Event("input", { bubbles: true }));
    return true;
  }
  // Contenteditable boxes (Claude, Gemini): select all, then insert,
  // which is the same thing your own paste would do.
  var sel = window.getSelection();
  var range = document.createRange();
  range.selectNodeContents(box);
  sel.removeAllRanges();
  sel.addRange(range);
  document.execCommand("insertText", false, text);
  return box.innerText.indexOf("clonemy.ai/extract") !== -1;
}

// ------------------------------------------------------------------
// Finding the self-report on the page. We score visible text blocks
// by how many of the 11 section titles (or numbered section headers)
// they contain, and keep the strongest one. Code blocks win ties:
// when an AI returns the report as one code block, its raw text is
// the cleanest copy.
// ------------------------------------------------------------------
// The extraction instructions themselves contain all 11 section
// titles, so when a user pastes them into a chat (the by-hand route,
// forced on Perplexity, live test 2026-10-31), the pasted template
// is the most report-shaped block on the page and the finder used to
// compile the questions instead of the answers. These marker lines
// appear in the instructions and never in a genuine report, so any
// block containing one is the template and is disqualified.
var TEMPLATE_MARKERS = [
  "Send this file to your current AI",
  "How to use (for you, the human)",
  "clonemy.ai extraction request in plain text"
];

function isTemplate(t) {
  for (var i = 0; i < TEMPLATE_MARKERS.length; i++) {
    if (t.indexOf(TEMPLATE_MARKERS[i]) !== -1) { return true; }
  }
  return false;
}

// Small cleanups on the winning block's text, each learned from a
// live test on 2026-10-31:
// - Code views (Claude's artifact in code mode) leak their line
//   numbers into the text as lines that are only digits. When a
//   text carries many of them (a code view signature), drop them.
// - Chat pages glue their own furniture to the tail of the last
//   section (AI mistake disclaimers, "file is ready" lines, stray
//   buttons). Trim known furniture lines from the very end only.
// - Some furniture lines are dropped wherever they appear
//   (CHROME_ANYWHERE). Learned from a live ChatGPT compile on
//   2026-11-01 (v0.1.6): the tail trim only pops while EVERY
//   trailing line matches, and one unmatched furniture line
//   ("Try Plus free") shielded the whole block above it, which
//   then landed inside section 10 of the finished capsule. The
//   anywhere list holds interface text that is never report
//   content, so dropping it on sight cannot eat a real line.
var CHROME_TAIL = [
  /can make mistakes/i,
  /is AI and can make mistakes/i,
  /^Edit directly or with Gemini$/i,
  /^The file .+ is ready for download\.?$/i,
  /^Latest response$/i,
  /^Think$/i,
  /^Document$/i,
  /^capsule-source\.md$/i
];

var CHROME_ANYWHERE = [
  /can make mistakes/i,
  /^Latest response$/i,
  /^Try Plus free$/i,
  /^Improve accuracy for documents and research$/i,
  /^Upgrade to Plus and use increased reasoning/i,
  /^capsule-source\.md$/i,
  // Live capture, v0.1.8 spot-check on ChatGPT, 2026-11-02: the
  // winning page block was the whole conversation column, whose
  // text runs past the report into the code block's language
  // label, the file chip, and a Pro upsell banner for "Astra"
  // that did not exist in the 2026-11-01 captures. Same shield
  // pattern as v0.1.6, new furniture. The chip's "Document" line
  // is promoted here from the tail list, and Perplexity's chip
  // reads "capsule-source" without the extension.
  /^Markdown$/i,
  /^Document$/i,
  /^capsule-source$/i,
  /Bring Astra to your most demanding work/i,
  /Upgrade to Pro for full access to Astra/i
];

function cleanReportText(t) {
  var lines = t.split("\n");
  var digitLines = 0;
  lines.forEach(function (ln) { if (/^\s*\d{1,4}\s*$/.test(ln)) { digitLines++; } });
  if (digitLines >= 5) {
    lines = lines.filter(function (ln) { return !/^\s*\d{1,4}\s*$/.test(ln); });
  }
  lines = lines.filter(function (ln) {
    var t = ln.trim();
    for (var i = 0; i < CHROME_ANYWHERE.length; i++) {
      if (CHROME_ANYWHERE[i].test(t)) { return false; }
    }
    return true;
  });
  while (lines.length) {
    var last = lines[lines.length - 1].trim();
    if (!last) { lines.pop(); continue; }
    var chrome = false;
    for (var i = 0; i < CHROME_TAIL.length; i++) {
      if (CHROME_TAIL[i].test(last)) { chrome = true; break; }
    }
    if (!chrome) { break; }
    lines.pop();
  }
  return lines.join("\n");
}

function scoreText(t) {
  if (!t || t.length < 200) { return 0; }
  var hits = 0;
  for (var i = 0; i < SECTION_TITLES.length; i++) {
    if (t.indexOf(SECTION_TITLES[i]) !== -1) { hits++; }
  }
  if (t.indexOf("About this report") === -1 && /(^|\n)\s{0,3}#{0,4}\s*0[.)]?\s+Meta\b/.test(t)) { hits++; }
  var numbered = (t.match(/(^|\n)\s{0,3}#{1,4}\s*\d{1,2}[.)]?\s/g) || []).length;
  return Math.max(hits, Math.min(numbered, 11));
}

function findReport() {
  var blocks = document.querySelectorAll("pre, code, [data-message-author-role], model-response, .markdown, .prose, article, main");
  var best = null, bestScore = 0, bestLen = 0, templateSeen = false;
  blocks.forEach(function (el) {
    var t = el.tagName === "PRE" || el.tagName === "CODE" ? el.textContent : el.innerText;
    if (!t) { return; }
    var s = scoreText(t);
    if (isTemplate(t)) {
      if (s >= 6) { templateSeen = true; }
      return;
    }
    if (s > bestScore || (s === bestScore && t.length > bestLen)) {
      best = t; bestScore = s; bestLen = t.length;
    }
  });
  if (bestScore >= 6) { return { text: cleanReportText(best), sections: bestScore }; }
  if (templateSeen) { return { template: true }; }
  return null;
}

// The strongest partial signal on the page: how many of the 11
// section headings the best non-template block holds so far.
// Used only for the status line, so a user waiting on a streaming
// answer can see the watcher counting instead of a grey button.
function partialScore() {
  var blocks = document.querySelectorAll("pre, code, [data-message-author-role], model-response, .markdown, .prose, article, main");
  var best = 0;
  blocks.forEach(function (el) {
    var t = el.tagName === "PRE" || el.tagName === "CODE" ? el.textContent : el.innerText;
    if (!t || isTemplate(t)) { return; }
    var s = scoreText(t);
    if (s > best) { best = s; }
  });
  return best;
}

function download(filename, text) {
  var blob = new Blob([text], { type: "text/markdown" });
  var url = URL.createObjectURL(blob);
  var a = document.createElement("a");
  a.href = url;
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  a.remove();
  setTimeout(function () { URL.revokeObjectURL(url); }, 2000);
}

function cmaiPageTheme() {
  try {
    var cs = window.getComputedStyle(document.documentElement).colorScheme || "";
    var hasLight = cs.indexOf("light") !== -1, hasDark = cs.indexOf("dark") !== -1;
    if (hasLight && !hasDark) { return "light"; }
    if (hasDark && !hasLight) { return "dark"; }
    if (hasLight && hasDark) {
      return window.matchMedia("(prefers-color-scheme: light)").matches ? "light" : "dark";
    }
  } catch (e) {}
  try {
    if (window.matchMedia("(prefers-color-scheme: light)").matches) { return "light"; }
    if (window.matchMedia("(prefers-color-scheme: dark)").matches) { return "dark"; }
  } catch (e) {}
  try {
    var bg = window.getComputedStyle(document.body || document.documentElement).backgroundColor;
    var m = /rgba?\((\d+),\s*(\d+),\s*(\d+)(?:,\s*([\d.]+))?\)/.exec(bg || "");
    if (m && m[4] !== "0") {
      var lum = (0.2126 * +m[1] + 0.7152 * +m[2] + 0.0722 * +m[3]) / 255;
      return lum > 0.6 ? "light" : "dark";
    }
  } catch (e) {}
  return "dark";
}

    var hit = findReport();

      // No finished report yet. If parts of one are on the page,
      // say so, with a count: a grey button and a silent bar read
      // as "broken" during the seconds the watcher needs, so the
      // bar narrates its own work while it works (Tony, live
      // Perplexity test, 2026-11-02: 5 to 10 seconds of dead air
      // between the answer finishing and Compile lighting up).
      if (!found) {
        var p = partialScore();
        if (p >= 1) {
          status.classList.add("cmai-scan");
          var msg = "Analyzing the answer as it arrives: " + p + " of 11 section headings spotted so far.";
          if (status.textContent !== msg) { status.textContent = msg; }
        }
      }
      return;
    }
    if (hit.template) {
      status.classList.remove("cmai-scan");
      if (!found && status.textContent !== TEMPLATE_NOTE) {
        status.textContent = TEMPLATE_NOTE;
      }
      return;
    }
    if (hit.text === lastText) {
      if (stableText !== hit.text) {
        stableText = hit.text;
        found = hit;
        status.classList.remove("cmai-scan");
        compileBtn.disabled = false;
        compileBtn.classList.add("cmai-ready");
        compileBtn.textContent = "2. Compile my capsule \u2193";
        var note = "Self-report found, " + hit.sections + " of 11 sections. Press Compile.";
        // On Claude, a report inside a document pane only shows
        // this watcher a window of the text at a time; the pane's
        // code view (</>) exposes the whole document at once.
        // Learned live, 2026-10-31: 8 of 11 in preview, 11 of 11
        // the moment the view switched.
        if (platform.id === "claude" && hit.sections < 11) {
          note += " If the report sits in a document pane, switch it to code view (</>) so the whole text is visible.";
        }
        status.textContent = note;
      }
    } else {
      lastText = hit.text;
      if (!found) {
        status.classList.add("cmai-scan");
        var waitMsg = "Report spotted. Making sure it has finished arriving...";
        if (status.textContent !== waitMsg) { status.textContent = waitMsg; }
      }
      // The page may go quiet right after the last chunk arrives,
      // with no further changes to observe. Check once more on a
      // timer so a finished report still counts as finished.
      setTimeout(scan, 1600);
    }
  }
  var observer = new MutationObserver(function () {
    clearTimeout(timer);
    timer = setTimeout(scan, 1200);
  });
  observer.observe(document.body, { childList: true, subtree: true, characterData: true });
}

// Start.
var platform = currentPlatform();
if (platform) {
  recordVisit(platform);
  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", function () { buildBar(platform); });
  } else {
    buildBar(platform);
  }
}

function b64e(bytes) {
  var s = "", CH = 0x8000;
  for (var i = 0; i < bytes.length; i += CH) { s += String.fromCharCode.apply(null, bytes.subarray(i, i + CH)); }
  return btoa(s);
}
function b64d(s) {
  var bin = atob(s), out = new Uint8Array(bin.length);
  for (var i = 0; i < bin.length; i++) { out[i] = bin.charCodeAt(i); }
  return out;
}
function deriveSync(pass, saltB64, iter, done) {
  crypto.subtle.importKey("raw", new TextEncoder().encode(pass), "PBKDF2", false, ["deriveKey"]).then(function (base) {
    return crypto.subtle.deriveKey({ name: "PBKDF2", salt: b64d(saltB64), iterations: iter, hash: "SHA-256" }, base, { name: "AES-GCM", length: 256 }, true, ["encrypt", "decrypt"]);
  }).then(function (key) {
    crypto.subtle.exportKey("raw", key).then(function (raw) {
      crypto.subtle.digest("SHA-256", raw).then(function (dg) {
        var check = b64e(new Uint8Array(dg).slice(0, 16));
        crypto.subtle.exportKey("jwk", key).then(function (jwk) { done({ jwk: jwk, salt: saltB64, iter: iter, check: check }); });
      });
    });
  }).catch(function () { done(null); });
}

# clonemy.ai browser extension

Move your AI companion's personality to any platform, without carrying files back and forth yourself. This extension lives inside your AI chat tab, helps run the extraction, captures the self-report where it was written, compiles your capsule locally, and downloads capsule.md.

Works on ChatGPT, Claude, Gemini, Grok, and Perplexity.

## The nothing-scary promise, and how to check it yourself

This extension was designed so that a person who reads basic code can verify there is nothing scary inside. Do not take our word for it. Check:

1. **Zero network requests.** Search every JavaScript file in this package for `fetch(`, `XMLHttpRequest`, `WebSocket`, `sendBeacon`. There are none. The extension has no code path that can transmit anything, so nothing you or your AI writes can be sent anywhere by it.
2. **No minification, no build step.** The files in the store package are these files, byte for byte, from the tagged release in this repository. Format is readable on purpose.
3. **Tiny permissions.** The manifest asks for exactly:
   - `storage`: three things in your browser, plus one optional fourth that leaves it only as ciphertext: a local tally of which AI sites you use (days and sessions, shown in the popup with an erase button), a capsule you compiled that is waiting to be delivered into another of your AI tabs (gone when you discard it from the bar, or after 24 hours), and your capsule vault: the latest capsule you compiled from each AI site, versioned so a recompile names the sections that changed, kept until you forget it from the bar, plus a record of which version each AI last received so updates can be offered instead of silent replacements. The optional fourth: if you turn on browser sync in the popup, encrypted chunks of the vault are written to your browser's sync storage and carried to your other browsers by the browser itself, protected by a six-word sync phrase the extension issues you (anyone who has the phrase can open the synced capsules, so keep it like a password); the sync carrier sees chunk sizes and timing, never content, and the extension itself makes no network requests.
   - Host access to the five AI sites above, so the bar can appear in those tabs and nowhere else.
   Nothing else. No history (we refuse that permission on principle), no cookies, no access to any other website.
4. **It never sends for you.** Button 1 puts the extraction request into the chat box. You press send. Button 2 compiles and downloads. Every consequential step is a press you make.

## How it works

1. Open your AI's chat. The clonemy bar sits in the corner.
2. Press "1. Put the request in the chat", then press send in the chat, like you normally would.
3. Your AI writes its self-report. When the extension recognizes the finished report (the numbered sections of the extraction format) it lights up button 2 with the section count.
4. Press "2. Compile my capsule". capsule.md downloads, compiled in the tab by compiler.js, a straight port of the compiler on clonemy.ai. Same input, same capsule.
5. Send capsule.md to your new AI with the line in its READ THIS FIRST header.

## Files

- `manifest.json`: what the browser enforces. Read this first.
- `content.js`: the bar, the fill, the watching, the download. Heavily commented.
- `compiler.js`: the capsule compiler. Pure text in, text out.
- `popup.html` / `popup.js`: the tally display and the regular-AI offer box.
- `icons/`: the clonemy.ai logo.

## Detection, in plain words

The extension counts, per AI site, how many sessions and how many different days you have used it, starting from install. It deliberately does not look at your history from before install; that permission is too broad to ask for. An AI used on 3 or more different days (or 5+ sessions) counts as your regular, and the popup then asks whether you want to sync it with clonemy.ai, transfer it, or sync it to another AI. The counts are visible in the popup and erasable there. They never leave your machine.

## Install for testing (no store needed)

Chrome or Edge: open `chrome://extensions`, turn on Developer mode, choose "Load unpacked", pick this folder.
